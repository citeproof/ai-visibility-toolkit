# robots.txt rules for AI crawlers
# Check each vendor's documentation for the current list of user agents.
#
# Search / answer crawlers: fetch pages so they can be cited in AI search answers.
# Blocking these makes it unlikely your pages are cited by that engine.
#
# Training crawlers: collect data for model training. You can block these
# and still allow the search crawlers.

# GPTBot: model training (change to Disallow: / to opt out)
User-agent: GPTBot
Allow: /

# --- Anthropic ---

Allow: /

# ClaudeBot: model training (change to Disallow: / to opt out)

# --- Perplexity ---

User-agent: *
